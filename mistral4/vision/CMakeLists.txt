# m4-vision: built inside a llama.cpp checkout's build (see mistral4/install.sh), linking its llama and mtmd targets.
cmake_minimum_required(VERSION 3.18)
project(m4_vision CXX)
set(LLAMA_CPP_DIR "" CACHE PATH "the llama.cpp source tree")
set(LLAMA_CPP_BUILD "" CACHE PATH "its build tree")
add_executable(m4-vision m4_vision.cpp)
target_compile_features(m4-vision PRIVATE cxx_std_17)
target_include_directories(m4-vision PRIVATE ${LLAMA_CPP_DIR}/include ${LLAMA_CPP_DIR}/tools/mtmd ${LLAMA_CPP_DIR}/ggml/include)
find_library(LIB_MTMD mtmd PATHS ${LLAMA_CPP_BUILD}/bin ${LLAMA_CPP_BUILD}/tools/mtmd ${LLAMA_CPP_BUILD}/lib NO_DEFAULT_PATH REQUIRED)
find_library(LIB_LLAMA llama PATHS ${LLAMA_CPP_BUILD}/bin ${LLAMA_CPP_BUILD}/src ${LLAMA_CPP_BUILD}/lib NO_DEFAULT_PATH REQUIRED)
find_library(LIB_GGML ggml PATHS ${LLAMA_CPP_BUILD}/bin ${LLAMA_CPP_BUILD}/ggml/src ${LLAMA_CPP_BUILD}/lib NO_DEFAULT_PATH REQUIRED)
find_library(LIB_GGML_BASE ggml-base PATHS ${LLAMA_CPP_BUILD}/bin ${LLAMA_CPP_BUILD}/ggml/src ${LLAMA_CPP_BUILD}/lib NO_DEFAULT_PATH REQUIRED)
target_link_libraries(m4-vision PRIVATE ${LIB_MTMD} ${LIB_LLAMA} ${LIB_GGML} ${LIB_GGML_BASE})
set_target_properties(m4-vision PROPERTIES BUILD_RPATH "${LLAMA_CPP_BUILD}/bin" INSTALL_RPATH "${LLAMA_CPP_BUILD}/bin")
